export const CopyBtn = ({ hex }) => (
  <button
    className="cp-copy-btn"
    type="button"
    onClick={(e) => {
      navigator.clipboard.writeText(hex);
      const b = e.currentTarget;
      b.classList.add("cp-copy-btn--copied");
      setTimeout(() => b.classList.remove("cp-copy-btn--copied"), 1500);
    }}
    title={`Copy ${hex}`}
    aria-label={`Copy ${hex}`}
  >
    <svg className="cp-copy-icon" width="14" height="14" viewBox="0 0 16 16" fill="none" aria-hidden="true">
      <rect x="5" y="5" width="9" height="9" rx="1.5" stroke="currentColor" strokeWidth="1.5" />
      <path d="M3 11V3a1 1 0 0 1 1-1h8" stroke="currentColor" strokeWidth="1.5" strokeLinecap="round" />
    </svg>
    <svg className="cp-check-icon" width="14" height="14" viewBox="0 0 16 16" fill="none" aria-hidden="true">
      <path d="M3 8l4 4 6-6" stroke="currentColor" strokeWidth="1.5" strokeLinecap="round" strokeLinejoin="round" />
    </svg>
  </button>
);

export const PaletteRow = ({ shades, tag = "Base", columns }) => (
  <div
    className="cp-palette-row"
    style={columns ? { gridTemplateColumns: `repeat(${columns}, minmax(0, 1fr))` } : undefined}
  >
    {shades.map(({ shade, hex, brand, base }) => {
      const isBase = Boolean(brand || base || shade === "600");
      const label = brand ? "Brand" : tag;
      return (
        <div
          key={shade}
          className={`cp-swatch-wrap${isBase ? " cp-swatch-wrap--base" : ""}`}
          style={isBase ? { "--cp-base": hex } : undefined}
        >
          <div className="cp-swatch" style={{ background: hex }}>
            <CopyBtn hex={hex} />
          </div>
          <span className="cp-swatch-shade">{shade}</span>
          <span className="cp-swatch-hex">{hex}</span>
          {isBase && <span className="cp-swatch-tag">{label}</span>}
        </div>
      );
    })}
  </div>
);

export const TokenTable = ({ title, description, rows }) => (

            <th>Usage</th>
