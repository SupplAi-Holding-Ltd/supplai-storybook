export const ButtonConfigurator = () => {
  const [variant, setVariant] = React.useState("filled");
  const [intent,  setIntent]  = React.useState("primary");
  const [size,    setSize]    = React.useState("md");
  const [withLeading,  setWithLeading]  = React.useState(false);
  const [withTrailing, setWithTrailing] = React.useState(false);
  const [isDisabled,   setIsDisabled]   = React.useState(false);
  const [isLoading,    setIsLoading]    = React.useState(false);

  const variants = ["filled", "outlined", "tonal", "ghost"];
  const intents  = ["primary", "success", "warning", "error", "neutral"];
  const sizes    = ["xs", "sm", "md", "lg", "xl"];

  const intentColors = {
    primary: "#4169E1", success: "#519E8A", warning: "#FF9F05",
    error: "#D13145", neutral: "#6B7280",
  };

  return (
    <div className="bc-wrap">
      {/* Preview */}
      <div className="bc-preview">
        <Button
          variant={variant}
          intent={intent}
          size={size}
          label={isLoading ? "Saving…" : "Button"}
          leadingIcon={withLeading ? <PlusIcon /> : undefined}
          trailingIcon={withTrailing ? <ArrowRightIcon /> : undefined}
          disabled={isDisabled}
          loading={isLoading}
        />
      </div>

      {/* Controls */}
      <div className="bc-controls">
        <div className="bc-control-group">
          <span className="bc-ctrl-label">Variant</span>
          <div className="bc-chips">
            {variants.map(v => (
              <button key={v} className={`bc-chip ${variant === v ? "bc-chip--on" : ""}`} onClick={() => setVariant(v)}>{v}</button>
            ))}
          </div>
        </div>

        <div className="bc-control-group">
          <span className="bc-ctrl-label">Intent</span>
          <div className="bc-chips">
            {intents.map(i => (
              <button
                key={i}
                className={`bc-chip ${intent === i ? "bc-chip--on" : ""}`}
                style={intent === i ? { background: intentColors[i], borderColor: intentColors[i], color: "#fff" } : {}}
                onClick={() => setIntent(i)}
              >{i}</button>
            ))}
          </div>
        </div>

        <div className="bc-control-group">
          <span className="bc-ctrl-label">Size</span>
          <div className="bc-chips">
            {sizes.map(s => (
              <button key={s} className={`bc-chip ${size === s ? "bc-chip--on" : ""}`} onClick={() => setSize(s)}>{s}</button>
            ))}
          </div>
        </div>

        <div className="bc-control-group">
          <span className="bc-ctrl-label">Options</span>
          <div className="bc-chips">
            <button className={`bc-chip ${withLeading  ? "bc-chip--on" : ""}`} onClick={() => setWithLeading(v => !v)}>Leading icon</button>
            <button className={`bc-chip ${withTrailing ? "bc-chip--on" : ""}`} onClick={() => setWithTrailing(v => !v)}>Trailing icon</button>
            <button className={`bc-chip ${isDisabled   ? "bc-chip--on" : ""}`} onClick={() => setIsDisabled(v => !v)}>Disabled</button>
            <button className={`bc-chip ${isLoading    ? "bc-chip--on" : ""}`} onClick={() => setIsLoading(v => !v)}>Loading</button>
          </div>
        </div>
      </div>
    </div>
  );
};

export const FormSubmitDemo = () => {
  const [email, setEmail] = React.useState("");
  const [password, setPassword] = React.useState("");
  const [error, setError] = React.useState("");
  const [loading, setLoading] = React.useState(false);
  const [done, setDone] = React.useState(false);

  const onSubmit = (event) => {
    event.preventDefault();
    if (!/^[^\s@]+@[^\s@]+\.[^\s@]+$/.test(email)) {
      setError("Enter a valid email address.");
      setDone(false);
      return;
    }
    setError("");
    setLoading(true);
    window.setTimeout(() => {
      setLoading(false);
      setDone(true);
    }, 900);
  };

  return (
    <form className="bd-form-card bd-form-card--signin" onSubmit={onSubmit}>
      <h4>Sign in</h4>
      <div className="bd-form-field">
        <label htmlFor="bd-email">Email</label>
        <input id="bd-email" className="bd-input" type="email" value={email} onChange={(e) => setEmail(e.target.value)} />
      </div>
      <div className="bd-form-field">
        <label htmlFor="bd-password">Password</label>
        <input id="bd-password" className="bd-input" type="password" value={password} onChange={(e) => setPassword(e.target.value)} />
      </div>
      {error ? <p className="bd-form-error" role="alert">{error}</p> : null}
      {done ? <p className="bd-form-ok" role="status">Signed in locally. No request was sent.</p> : null}
      <div className="bd-form-actions">
        <Button type="button" variant="outlined" intent="neutral" label="Cancel" disabled={loading} onClick={() => { setEmail(""); setPassword(""); setError(""); setDone(false); }} />
        <Button type="submit" variant="filled" intent="primary" label={loading ? "Signing in…" : "Sign in"} loading={loading} />
      </div>
    </form>
  );
};
