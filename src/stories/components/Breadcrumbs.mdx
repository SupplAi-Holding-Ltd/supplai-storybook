export const BREADCRUMB_CODE = `// Breadcrumb · supplai Design System
import React from 'react';

function Breadcrumb({ children }) {
  return <nav aria-label="Breadcrumb">{children}</nav>;
}

function BreadcrumbList({ children }) {
  return <ol className="bc-list">{children}</ol>;
}

function BreadcrumbItem({ children }) {
  return <li className="bc-item">{children}</li>;
}

function BreadcrumbLink({ href = '#', children }) {
  return <a className="bc-link" href={href}>{children}</a>;
}

function BreadcrumbPage({ children }) {
  return <span className="bc-current" aria-current="page">{children}</span>;
}

function BreadcrumbSeparator({ children }) {
  return (
    <li className="bc-sep" aria-hidden="true">
      {children ?? <span className="bc-chevron" />}
    </li>
  );
}

function BreadcrumbEllipsis() {
  return <span className="bc-ellipsis" aria-label="More">⋯</span>;
}
`;

export const Chevron = () => (
  <svg width="14" height="14" viewBox="0 0 16 16" fill="none" aria-hidden="true">
    <path
      d="M6 4l4 4-4 4"
      stroke="currentColor"
      strokeWidth="1.5"
      strokeLinecap="round"
      strokeLinejoin="round"
    />
  </svg>
);

export const Breadcrumb = ({ children, className = "" }) => (
  <nav aria-label="Breadcrumb" className={`bc${className ? ` ${className}` : ""}`}>
    {children}
  </nav>
);

export const BreadcrumbList = ({ children, className = "" }) => (
  <ol className={`bc-list${className ? ` ${className}` : ""}`}>{children}</ol>
);

export const BreadcrumbItem = ({ children, className = "" }) => (
  <li className={`bc-item${className ? ` ${className}` : ""}`}>{children}</li>
);

export const BreadcrumbLink = ({ href = "#", children, className = "", onClick }) => (
  <a
    className={`bc-link${className ? ` ${className}` : ""}`}
    href={href}
    onClick={(e) => {
      e.preventDefault();
      onClick?.(e);
    }}
  >
    {children}
  </a>
);

export const BreadcrumbPage = ({ children, className = "" }) => (
  <span className={`bc-current${className ? ` ${className}` : ""}`} aria-current="page">
    {children}
  </span>
);

export const BreadcrumbSeparator = ({ children, className = "" }) => (
  <li className={`bc-sep${className ? ` ${className}` : ""}`} aria-hidden="true" role="presentation">
    {children ?? <Chevron />}
  </li>
);

export const BreadcrumbEllipsis = ({ className = "" }) => (
  <span className={`bc-ellipsis${className ? ` ${className}` : ""}`} aria-label="More pages">
    ···
  </span>
);

export const BreadcrumbPlayground = () => {
  const [separator, setSeparator] = React.useState("chevron");
  const [withIcons, setWithIcons] = React.useState(false);
  const [collapsed, setCollapsed] = React.useState(true);
  const [card, setCard] = React.useState(false);

  const Sep = () => {
    if (separator === "slash") return <BreadcrumbSeparator>/</BreadcrumbSeparator>;
    if (separator === "dot") return <BreadcrumbSeparator><span className="bc-dot-sep" /></BreadcrumbSeparator>;
    return <BreadcrumbSeparator />;
  };

  return (
    <div className="bcp-wrap">
      <div className="bcp-preview">
        <Breadcrumb className={card ? "bc--card" : ""}>
          <BreadcrumbList>
            <BreadcrumbItem>
              <BreadcrumbLink href="/">
                {withIcons && <Door size={14} />}
                Home
              </BreadcrumbLink>
            </BreadcrumbItem>
            <Sep />
            {collapsed ? (
              <>
                <BreadcrumbItem>
                  <BreadcrumbEllipsis />
                </BreadcrumbItem>
                <Sep />
              </>
            ) : (
              <>
                <BreadcrumbItem>
                  <BreadcrumbLink href="/docs">
                    {withIcons && <Folder size={14} />}
                    Docs
                  </BreadcrumbLink>
                </BreadcrumbItem>
                <Sep />
              </>
            )}
            <BreadcrumbItem>
              <BreadcrumbLink href="/docs/components">
                {withIcons && <Folder size={14} />}
                Components
              </BreadcrumbLink>
            </BreadcrumbItem>
            <Sep />
            <BreadcrumbItem>
              <BreadcrumbPage>
                {withIcons && <File size={14} />}
                Breadcrumb
              </BreadcrumbPage>
            </BreadcrumbItem>
          </BreadcrumbList>
        </Breadcrumb>
      </div>
      <div className="bcp-controls">
        <div className="bcp-ctrl-group">
          <span className="bcp-ctrl-label">Separator</span>
          <div className="bcp-chips">
            {["chevron", "slash", "dot"].map((v) => (
              <button
                key={v}
                type="button"
                className={`bcp-chip ${separator === v ? "bcp-chip--on" : ""}`}
                onClick={() => setSeparator(v)}
              >
                {v}
              </button>
            ))}
          </div>
        </div>
        <div className="bcp-ctrl-group">
          <span className="bcp-ctrl-label">Options</span>
          <div className="bcp-chips">
            <button
              type="button"
              className={`bcp-chip ${withIcons ? "bcp-chip--on" : ""}`}
              onClick={() => setWithIcons((v) => !v)}
            >
              Icons
            </button>
            <button
              type="button"
              className={`bcp-chip ${collapsed ? "bcp-chip--on" : ""}`}
              onClick={() => setCollapsed((v) => !v)}
            >
              Ellipsis
            </button>
            <button
              type="button"
              className={`bcp-chip ${card ? "bcp-chip--on" : ""}`}
              onClick={() => setCard((v) => !v)}
            >
              Card
            </button>
          </div>
        </div>
      </div>
    </div>
  );
};

export const DropdownBreadcrumb = () => {
  const [open, setOpen] = React.useState(false);
  const ref = React.useRef(null);

  React.useEffect(() => {
    const onDoc = (e) => {
      if (ref.current && !ref.current.contains(e.target)) setOpen(false);
    };
    document.addEventListener("mousedown", onDoc);
    return () => document.removeEventListener("mousedown", onDoc);
  }, []);

  return (
    <Breadcrumb>
      <BreadcrumbList>
        <BreadcrumbItem>
          <BreadcrumbLink href="/">Home</BreadcrumbLink>
        </BreadcrumbItem>
        <BreadcrumbSeparator />
        <BreadcrumbItem>
          <div className="bc-dd" ref={ref}>
            <button
              type="button"
              className="bc-dd-trigger"
              aria-expanded={open}
              aria-haspopup="menu"
              onClick={() => setOpen((v) => !v)}
            >
              <BreadcrumbEllipsis />
            </button>
            {open && (
              <ul className="bc-dd-menu" role="menu">
                {["Getting Started", "Installation", "Fundamentals"].map((label) => (
                  <li key={label} role="none">
                    <button
                      type="button"
                      role="menuitem"
                      className="bc-dd-item"
                      onClick={() => setOpen(false)}
                    >
                      {label}
                    </button>
                  </li>
                ))}
              </ul>
            )}
          </div>
        </BreadcrumbItem>
        <BreadcrumbSeparator />
        <BreadcrumbItem>
          <BreadcrumbLink href="/components">Components</BreadcrumbLink>
        </BreadcrumbItem>
        <BreadcrumbSeparator />
        <BreadcrumbItem>
          <BreadcrumbPage>Toast</BreadcrumbPage>
        </BreadcrumbItem>
      </BreadcrumbList>
    </Breadcrumb>
  );
};
