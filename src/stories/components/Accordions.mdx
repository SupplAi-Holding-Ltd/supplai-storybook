import React, { useState } from 'react';

// variant:   box | table | open
// indicator: chevron | plus-minus
// type:      single | multiple

function Accordion({
  type = 'single',
  collapsible = true,
  variant = 'box',
  indicator = 'chevron',
  defaultValue,
  children,
}) {
  const [open, setOpen] = useState(() => {
    if (defaultValue == null) return type === 'multiple' ? [] : null;
    return type === 'multiple'
      ? (Array.isArray(defaultValue) ? defaultValue : [defaultValue])
      : defaultValue;
  });

  const isOpen = (value) =>
    type === 'multiple' ? open.includes(value) : open === value;

  const toggle = (value) => {
    if (type === 'multiple') {
      setOpen((prev) =>
        prev.includes(value) ? prev.filter((v) => v !== value) : [...prev, value]
      );
      return;
    }
    setOpen((prev) => (prev === value && collapsible ? null : value));
  };

  return (
    <div className={\`ac ac--\${variant}\`} data-orientation="vertical">
      {React.Children.map(children, (child) =>
        React.isValidElement(child)
          ? React.cloneElement(child, {
              open: isOpen(child.props.value),
              onToggle: () => toggle(child.props.value),
              indicator,
              variant,
            })
          : child
      )}
    </div>
  );
}

function AccordionItem({ value, disabled, open, onToggle, indicator, variant, children }) {
  return (
    <div
      className={\`ac-item\${open ? ' ac-item--open' : ''}\${disabled ? ' ac-item--disabled' : ''}\`}
      data-state={open ? 'open' : 'closed'}
    >
      {React.Children.map(children, (child) =>
        React.isValidElement(child)
          ? React.cloneElement(child, { open, onToggle, disabled, indicator, variant })
          : child
      )}
    </div>
  );
}

function AccordionTrigger({ children, open, onToggle, disabled, indicator, leading }) {
  return (
    <button
      type="button"
      className="ac-trigger"
      aria-expanded={!!open}
      disabled={disabled}
      onClick={() => !disabled && onToggle?.()}
    >
      {leading}
      <span className="ac-trigger-label">{children}</span>
      <span className={\`ac-indicator ac-indicator--\${indicator}\${open ? ' ac-indicator--open' : ''}\`} aria-hidden="true" />
    </button>
  );
}

function AccordionContent({ children, open }) {
  return (
    <div className={\`ac-content\${open ? ' ac-content--open' : ''}\`} aria-hidden={!open}>
      <div className="ac-content-inner">{children}</div>
    </div>
  );
}
`;

export const Accordion = ({
  type = "single",
  collapsible = true,
  variant = "box",
  indicator = "chevron",
  defaultValue,
  value: controlledValue,
  onValueChange,
  children,
  className = "",
}) => {
  const [uncontrolled, setUncontrolled] = React.useState(() => {
    if (defaultValue == null) return type === "multiple" ? [] : null;
    return type === "multiple"
      ? Array.isArray(defaultValue)
        ? defaultValue
        : [defaultValue]
      : defaultValue;
  });

  const openState = controlledValue !== undefined ? controlledValue : uncontrolled;

  const isOpen = (value) =>
    type === "multiple"
      ? Array.isArray(openState) && openState.includes(value)
      : openState === value;

  const toggle = (value) => {
    let next;
    if (type === "multiple") {
      const prev = Array.isArray(openState) ? openState : [];
      next = prev.includes(value) ? prev.filter((v) => v !== value) : [...prev, value];
    } else {
      next = openState === value && collapsible ? null : value;
    }
    if (controlledValue === undefined) setUncontrolled(next);
    onValueChange?.(next);
  };

  return (
    <div
      className={`ac ac--${variant}${className ? ` ${className}` : ""}`}
      data-orientation="vertical"
    >
      {React.Children.map(children, (child) =>
        React.isValidElement(child)
          ? React.cloneElement(child, {
              open: isOpen(child.props.value),
              onToggle: () => toggle(child.props.value),
              indicator,
              variant,
            })
          : child
      )}
    </div>
  );
};

export const AccordionItem = ({
  value,
  disabled = false,
  open,
  onToggle,
  indicator,
  variant,
  children,
  className = "",
}) => (
  <div
    className={`ac-item${open ? " ac-item--open" : ""}${disabled ? " ac-item--disabled" : ""}${className ? ` ${className}` : ""}`}
    data-state={open ? "open" : "closed"}
  >
    {React.Children.map(children, (child) =>
      React.isValidElement(child)
        ? React.cloneElement(child, { open, onToggle, disabled, indicator, variant })
        : child
    )}
  </div>
);

export const AccordionTrigger = ({
  children,
  open,
  onToggle,
  disabled,
  indicator = "chevron",
  leading,
  trailing,
  chevronSide = "right",
  className = "",
}) => {
  const indicatorNode =
    indicator === "plus-minus" ? (
      <PlusMinusIcon open={!!open} />
    ) : (
      <ChevronIcon open={!!open} />
    );

  return (
    <button
      type="button"
      className={`ac-trigger ac-trigger--chevron-${chevronSide}${className ? ` ${className}` : ""}`}
      aria-expanded={!!open}
      disabled={disabled}
      onClick={() => !disabled && onToggle?.()}
    >
      {chevronSide === "left" && (
        <span className="ac-indicator" aria-hidden="true">
          {indicatorNode}
        </span>
      )}
      {leading && <span className="ac-leading">{leading}</span>}
      <span className="ac-trigger-body">{children}</span>
      {trailing && <span className="ac-trailing">{trailing}</span>}
      {chevronSide === "right" && (
        <span className="ac-indicator" aria-hidden="true">
          {indicatorNode}
        </span>
      )}
    </button>
  );
};

export const AccordionContent = ({ children, open, className = "" }) => (

    role="region"
    aria-hidden={!open}
  >
    <div className="ac-content-inner">{children}</div>
  </div>
);
