export const AVATAR_CODE = `// Avatar · supplai Design System
import React, { useState } from 'react';

const SIZES = {
  16: { px: 16, font: 8,  status: 6,  badge: 10 },
  20: { px: 20, font: 9,  status: 6,  badge: 12 },
  24: { px: 24, font: 10, status: 7,  badge: 12 },
  32: { px: 32, font: 12, status: 8,  badge: 14 },
  40: { px: 40, font: 14, status: 10, badge: 16 },
  48: { px: 48, font: 16, status: 12, badge: 18 },
  56: { px: 56, font: 18, status: 14, badge: 20 },
  64: { px: 64, font: 20, status: 14, badge: 22 },
  80: { px: 80, font: 24, status: 16, badge: 24 },
};

const STATUS = {
  online:  '#519E8A',
  offline: '#94A3B8',
  busy:    '#D13145',
  away:    '#FF9F05',
};

function Avatar({
  size = 40,
  rounded = 'circle',
  src,
  alt = '',
  fallback = '?',
  status,
  badge,
  ring = false,
}) {
  const [failed, setFailed] = useState(false);
  const s = SIZES[size] || SIZES[40];
  const radius = rounded === 'square' ? Math.max(4, s.px * 0.2) : '50%';
  const showImage = src && !failed;

  return (
    <span
      className="av-root"
      style={{
        width: s.px,
        height: s.px,
        borderRadius: radius,
        boxShadow: ring ? '0 0 0 2px #fff, 0 0 0 4px #4169E1' : undefined,
      }}
    >
      {showImage ? (
        <img src={src} alt={alt} onError={() => setFailed(true)} />
      ) : (
        <span className="av-fallback" style={{ fontSize: s.font }}>
          {fallback}
        </span>
      )}
      {status && (
        <span
          className="av-status"
          style={{
            width: s.status,
            height: s.status,
            background: STATUS[status],
          }}
        />
      )}
      {badge != null && (
        <span className="av-badge" style={{ minWidth: s.badge, height: s.badge, fontSize: Math.max(9, s.badge * 0.55) }}>
          {badge}
        </span>
      )}
    </span>
  );
}

function AvatarGroup({ children, size = 40, max }) {
  const items = React.Children.toArray(children);
  const visible = max != null ? items.slice(0, max) : items;
  const overflow = max != null ? Math.max(0, items.length - max) : 0;
  return (
    <div className="av-group">
      {visible}
      {overflow > 0 && (
        <Avatar size={size} fallback={\`+\${overflow}\`} />
      )}
    </div>
  );
}
`;

export const Avatar = ({
  size = 40,
  rounded = "circle",
  src,
  alt = "",
  fallback = "?",
  fallbackIcon = false,
  status,
  badge,
  ring = false,
  bg = "#4169E1",
}) => {
  const [failed, setFailed] = React.useState(false);
  const s = SIZES[size] || SIZES[40];
  const radius = rounded === "square" ? Math.max(4, Math.round(s.px * 0.2)) : "50%";
  const showImage = Boolean(src) && !failed;

  return (
    <span
      className={`av-root${ring ? " av-root--ring" : ""}`}
      style={{
        width: s.px,
        height: s.px,
        borderRadius: radius,
        background: showImage ? "#E2E8F0" : bg,
      }}
      role="img"
      aria-label={alt || (typeof fallback === "string" ? fallback : "Avatar")}
    >
      {showImage ? (
        <img src={src} alt="" onError={() => setFailed(true)} />
      ) : fallbackIcon ? (
        <span className="av-fallback av-fallback--icon">
          <User size={Math.max(12, Math.round(s.px * 0.45))} />
        </span>
      ) : (
        <span className="av-fallback" style={{ fontSize: s.font }}>
          {fallback}
        </span>
      )}
      {status && (
        <span
          className="av-status"
          style={{
            width: s.status,
            height: s.status,
            background: STATUS_COLORS[status] || STATUS_COLORS.online,
            borderRadius: "50%",
          }}
          title={status}
        />
      )}
      {badge != null && badge !== "" && (
        <span
          className="av-badge"
          style={{
            minWidth: s.badge,
            height: s.badge,
            fontSize: Math.max(9, Math.round(s.badge * 0.55)),
            borderRadius: 999,
          }}
        >
          {badge}
        </span>
      )}
    </span>
  );
};
