export const Banner = ({
  variant = "soft",
  color = "primary",
  title,
  description,
  icon,
  action,
  onClose,
  floating = false,
  promo = false,
  children,
}) => {
  const c = COLORS[color] || COLORS.primary;
  const isStrong = variant === "strong";
  const isOutline = variant === "outline" || variant === "soft-outline";
  const softOutline = variant === "soft-outline";

  let background = "transparent";
  let foreground = c.softFg;
  let muted = c.softMuted;
  let borderColor = "transparent";

  if (isStrong) {
    background = c.strongBg;
    foreground = c.strongFg;
    muted = c.strongFg;
    borderColor = c.strongBg;
  } else if (variant === "soft") {
    background = c.softBg;
    foreground = c.softFg;
    muted = c.softMuted;
    borderColor = c.softBg;
  } else if (variant === "outline") {
    background = "#FFFFFF";
    foreground = c.softFg;
    muted = c.softMuted;
    borderColor = c.border;
  } else if (softOutline) {
    background = c.softBg;
    foreground = c.softFg;
    muted = c.softMuted;
    borderColor = c.border;
  }

  if (promo) {
    background = "linear-gradient(90deg, #0D1B2A 0%, #2F50C1 55%, #4169E1 100%)";
    foreground = "#FFFFFF";
    muted = "rgba(255,255,255,0.82)";
    borderColor = "transparent";
  }

  return (
    <div
      role={color === "error" || color === "warning" ? "alert" : "status"}
      className={`bn${floating ? " bn--floating" : ""}${promo ? " bn--promo" : ""}`}
      style={{
        background,
        color: foreground,
        borderColor,
        borderStyle: "solid",
        borderWidth: isOutline || softOutline || floating ? 1 : 0,
      }}
    >
      <div className="bn-content">
        {icon && (
          <span className="bn-icon" style={{ color: isStrong || promo ? foreground : muted }}>
            {icon}
          </span>
        )}
        <div className="bn-text">
          {title && <p className="bn-title" style={{ color: foreground }}>{title}</p>}
          {description && (
            <p className="bn-desc" style={{ color: isStrong || promo ? muted : muted, opacity: isStrong || promo ? 0.9 : 1 }}>
              {description}
            </p>
          )}
          {children}
        </div>
        {action && <div className="bn-action">{action}</div>}
      </div>
      {onClose && (
        <button
          type="button"
          className="bn-close"
          onClick={onClose}
          aria-label="Dismiss banner"
          style={{ color: isStrong || promo ? foreground : muted }}
        >
          <Cross size={16} />
        </button>
      )}
    </div>
  );
};

export const BnButton = ({ children, strong = false, color = "primary", onClick }) => {
  const c = COLORS[color] || COLORS.primary;
  return (
    <button
      type="button"
      className="bn-btn"
      onClick={onClick}
      style={
        strong
          ? { background: "#FFFFFF", color: c.strongBg, borderColor: "#FFFFFF" }
          : { background: "transparent", color: "inherit", borderColor: "currentColor" }
      }
    >
      {children}
    </button>
  );
};

export const BannerPlayground = () => {
  const [variant, setVariant] = React.useState("soft");
  const [color, setColor] = React.useState("primary");
  const [dismissible, setDismissible] = React.useState(true);
  const [visible, setVisible] = React.useState(true);
  const [withIcon, setWithIcon] = React.useState(true);
  const [withAction, setWithAction] = React.useState(false);

  const icons = {
    primary: <Bulb size={18} />,
    neutral: <Bell size={18} />,
    success: <Shield size={18} />,
    warning: <Warning size={18} />,
    error: <Flag size={18} />,
    info: <InfoCircle size={18} />,
  };

  return (
    <div className="bnp-wrap">
      <div className="bnp-preview">
        {visible ? (
          <Banner
            variant={variant}
            color={color}
            title="New Feature!"
            description="Check it out in the design system docs."
            icon={withIcon ? icons[color] : undefined}
            action={
              withAction ? (
                <BnButton strong={variant === "strong"} color={color}>
                  Learn more
                </BnButton>
              ) : undefined
            }
            onClose={dismissible ? () => setVisible(false) : undefined}
          />
        ) : (
          <button type="button" className="bnp-restore" onClick={() => setVisible(true)}>
            Show banner again
          </button>
        )}
      </div>
      <div className="bnp-controls">
        <div className="bnp-ctrl-group">
          <span className="bnp-ctrl-label">Variant</span>
          <div className="bnp-chips">
            {["soft", "strong", "outline", "soft-outline"].map((v) => (
              <button
                key={v}
                type="button"
                className={`bnp-chip ${variant === v ? "bnp-chip--on" : ""}`}
                onClick={() => setVariant(v)}
              >
                {v}
              </button>
            ))}
          </div>
        </div>
        <div className="bnp-ctrl-group">
          <span className="bnp-ctrl-label">Color</span>
          <div className="bnp-chips">
            {["primary", "neutral", "success", "warning", "error", "info"].map((c) => (
              <button
                key={c}
                type="button"
                className={`bnp-chip ${color === c ? "bnp-chip--on" : ""}`}
                onClick={() => setColor(c)}
              >
                {c}
              </button>
            ))}
          </div>
        </div>
        <div className="bnp-ctrl-group">
          <span className="bnp-ctrl-label">Options</span>
          <div className="bnp-chips">
            <button
              type="button"
              className={`bnp-chip ${withIcon ? "bnp-chip--on" : ""}`}
              onClick={() => setWithIcon((v) => !v)}
            >
              Icon
            </button>
            <button
              type="button"
              className={`bnp-chip ${withAction ? "bnp-chip--on" : ""}`}
              onClick={() => setWithAction((v) => !v)}
            >
              Action
            </button>
            <button
              type="button"
              className={`bnp-chip ${dismissible ? "bnp-chip--on" : ""}`}
              onClick={() => {
                setDismissible((v) => !v);
                setVisible(true);
              }}
            >
              Dismiss
            </button>
          </div>
        </div>
      </div>
    </div>
  );
};
