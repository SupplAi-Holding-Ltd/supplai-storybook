}

export const SIZE_MAP = {
  20: { h: 20, padX: 8, font: 9, gap: 4, icon: 11, avatar: 14, radius: 6 },
  24: { h: 24, padX: 10, font: 10, gap: 5, icon: 13, avatar: 16, radius: 7 },
  28: { h: 28, padX: 12, font: 11, gap: 6, icon: 14, avatar: 18, radius: 8 },

export const Badge = ({
  variant = "soft",
  color = "primary",
  size = 24,
  dot = false,
  rounded = "default",
  onRemove,
  leading,
  trailing,
  children,
  className = "",
  style,
  onClick,
  type,
  as: As = "span",
}) => {
  const c = BADGE_COLORS[color] || BADGE_COLORS.primary;
  const s = SIZE_MAP[size] || SIZE_MAP[24];
  const isStrong = variant === "strong";
  const isOutline = variant === "outline";

  let background = c.softBg;
  let foreground = c.softFg;
  let borderColor = "transparent";

  if (isStrong) {
    background = c.strongBg;
    foreground = c.strongFg;
    borderColor = c.strongBg;
  } else if (isOutline) {
    background = "#FFFFFF";
    foreground = c.softFg;
    borderColor = c.border;
  }

  const radius = rounded === "pill" ? 999 : s.radius;

  return (
    <As
      type={As === "button" ? type || "button" : undefined}
      className={`bdg${onClick ? " bdg--interactive" : ""}${className ? ` ${className}` : ""}`}
      onClick={onClick}
      style={{
        height: s.h,
        padding: `0 ${s.padX}px`,
        gap: s.gap,
        fontSize: s.font,
        borderRadius: radius,
        background,
        color: foreground,
        borderColor,
        borderStyle: "solid",
        borderWidth: isOutline ? 1 : 0,
        cursor: onClick ? "pointer" : undefined,
        ["--bdg-dot"]: c.dot,
        ...style,
      }}
    >
      {dot && <span className="bdg-dot" aria-hidden="true" />}
      {leading && <span className="bdg-lead">{leading}</span>}
      <span className="bdg-label">{children}</span>
      {trailing && <span className="bdg-trail">{trailing}</span>}
      {onRemove && (
        <button
          type="button"
          className="bdg-remove"
          onClick={(e) => {
            e.stopPropagation();
            onRemove(e);
          }}
          aria-label="Remove"
        >
          <Cross size={Math.max(10, s.icon - 2)} />
        </button>
      )}
    </As>
  );
};

export const RemovableRow = () => {
  const [tags, setTags] = React.useState(["VIP", "New Customer", "Priority"]);
  return (
    <div className="bdg-row">
      {tags.map((t) => (
        <Badge
          key={t}
          variant="soft"
          color="primary"
          size={24}
          rounded="pill"
          onRemove={() => setTags((prev) => prev.filter((x) => x !== t))}
        >
          {t}
        </Badge>
      ))}
      <button
        type="button"
        className="bdg-add"
        onClick={() =>
          setTags((prev) => (prev.includes("Partner") ? prev : [...prev, "Partner"]))
        }
      >
        + Add
      </button>
    </div>
  );
};

export const InteractiveBadges = () => {
  const [active, setActive] = React.useState(["Required", "Verified"]);
  const options = ["Required", "Metrics", "Invited", "Verified"];
  return (
    <div className="bdg-row">
      {options.map((label) => {
        const on = active.includes(label);
        return (
          <Badge
            key={label}
            as="button"
            type="button"
            variant={on ? "strong" : "outline"}
            color={on ? "primary" : "neutral"}
            size={24}
            rounded="pill"
            onClick={() =>
              setActive((prev) =>
                prev.includes(label) ? prev.filter((x) => x !== label) : [...prev, label]
              )
            }
          >
            {label}
          </Badge>
        );
      })}
    </div>
  );
};
