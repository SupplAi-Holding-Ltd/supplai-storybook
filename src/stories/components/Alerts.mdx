export const Alert = ({
  variant = "soft",
  color = "primary",
  children,
  onClose,
  className = "",
}) => {
  const c = COLORS[color] || COLORS.primary;
  const isStrong = variant === "strong";
  const isOutline = variant === "outline" || variant === "soft-outline";
  const softOutline = variant === "soft-outline";

  let background = "transparent";
  let foreground = c.softFg;
  let muted = c.softMuted;
  let borderColor = "transparent";

  if (isStrong) {
    background = c.strongBg;
    foreground = c.strongFg;
    muted = c.strongFg;
    borderColor = c.strongBg;
  } else if (variant === "soft") {
    background = c.softBg;
    foreground = c.softFg;
    muted = c.softMuted;
    borderColor = c.softBg;
  } else if (variant === "outline") {
    background = "#FFFFFF";
    foreground = c.softFg;
    muted = c.softMuted;
    borderColor = c.border;
  } else if (softOutline) {
    background = c.softBg;
    foreground = c.softFg;
    muted = c.softMuted;
    borderColor = c.border;
  }

  return (
    <div
      role={color === "error" || color === "warning" ? "alert" : "status"}
      className={`al al--${variant}${className ? ` ${className}` : ""}`}
      style={{
        background,
        color: foreground,
        borderColor,
        borderStyle: "solid",
        borderWidth: isOutline || softOutline ? 1 : 0,
        ["--al-fg"]: foreground,
        ["--al-muted"]: muted,
      }}
    >
      <div className="al-body">{children}</div>
      {onClose && (
        <button
          type="button"
          className="al-close"
          onClick={onClose}
          aria-label="Dismiss alert"
          style={{ color: isStrong ? foreground : muted }}
        >
          <Cross size={16} />
        </button>
      )}
    </div>
  );
};

export const AlButton = ({ children, strong = false, color = "primary", onClick, ghost = false }) => {
  const c = COLORS[color] || COLORS.primary;
  if (ghost) {
    return (
      <button type="button" className="al-btn al-btn--ghost" onClick={onClick}>
        {children}
      </button>
    );
  }
  return (
    <button
      type="button"
      className="al-btn"
      onClick={onClick}
      style={
        strong
          ? { background: "#FFFFFF", color: c.strongBg, borderColor: "#FFFFFF" }
          : { background: "transparent", color: "inherit", borderColor: "currentColor" }
      }
    >
      {children}
    </button>
  );
};

export const AlertPlayground = () => {
  const [variant, setVariant] = React.useState("soft");
  const [color, setColor] = React.useState("primary");
  const [dismissible, setDismissible] = React.useState(true);
  const [visible, setVisible] = React.useState(true);
  const [withIcon, setWithIcon] = React.useState(true);
  const [withAction, setWithAction] = React.useState(true);

  const icons = {
    primary: <Bulb size={18} />,
    neutral: <Bell size={18} />,
    success: <Shield size={18} />,
    warning: <Warning size={18} />,
    error: <Flag size={18} />,
    info: <InfoCircle size={18} />,
  };

  return (
    <div className="alp-wrap">
      <div className="alp-preview">
        {visible ? (
          <Alert
            variant={variant}
            color={color}
            onClose={dismissible ? () => setVisible(false) : undefined}
          >
            {withIcon && <AlertIcon>{icons[color]}</AlertIcon>}
            <AlertContent>
              <AlertTitle>Alert Title</AlertTitle>
              <AlertDescription>This is alert description.</AlertDescription>
            </AlertContent>
            {withAction && (
              <AlertToolbar>
                <AlButton strong={variant === "strong"} color={color}>
                  Action
                </AlButton>
              </AlertToolbar>
            )}
          </Alert>
        ) : (
          <button type="button" className="alp-restore" onClick={() => setVisible(true)}>
            Show alert again
          </button>
        )}
      </div>
      <div className="alp-controls">
        <div className="alp-ctrl-group">
          <span className="alp-ctrl-label">Variant</span>
          <div className="alp-chips">
            {["soft", "soft-outline", "outline", "strong"].map((v) => (
              <button
                key={v}
                type="button"
                className={`alp-chip ${variant === v ? "alp-chip--on" : ""}`}
                onClick={() => setVariant(v)}
              >
                {v}
              </button>
            ))}
          </div>
        </div>
        <div className="alp-ctrl-group">
          <span className="alp-ctrl-label">Color</span>
          <div className="alp-chips">
            {["primary", "neutral", "info", "success", "warning", "error"].map((c) => (
              <button
                key={c}
                type="button"
                className={`alp-chip ${color === c ? "alp-chip--on" : ""}`}
                onClick={() => setColor(c)}
              >
                {c}
              </button>
            ))}
          </div>
        </div>
        <div className="alp-ctrl-group">
          <span className="alp-ctrl-label">Options</span>
          <div className="alp-chips">
            <button
              type="button"
              className={`alp-chip ${withIcon ? "alp-chip--on" : ""}`}
              onClick={() => setWithIcon((v) => !v)}
            >
              Icon
            </button>
            <button
              type="button"
              className={`alp-chip ${withAction ? "alp-chip--on" : ""}`}
              onClick={() => setWithAction((v) => !v)}
            >
              Action
            </button>
            <button
              type="button"
              className={`alp-chip ${dismissible ? "alp-chip--on" : ""}`}
              onClick={() => {
                setDismissible((v) => !v);
                setVisible(true);
              }}
            >
              Dismiss
            </button>
          </div>
        </div>
      </div>
    </div>
  );
};

    </Alert>
  );
};
